========================================
QUANTA CLI COMMANDS CHEAT SHEET
========================================

1. BUILDING
-----------
cargo build --release


2. RUNNING NODE
---------------
# Start node
./target/release/quanta start

# Start with custom config
./target/release/quanta start -c quanta.toml

# Start in detached mode (background)
./target/release/quanta start --detach


3. WALLET OPERATIONS
--------------------
# Create new wallet
./target/release/quanta new-wallet --file wallet.qua

# View wallet details
./target/release/quanta wallet --file wallet.qua

# Get balance
curl -X POST http://localhost:3000/api/balance -H "Content-Type: application/json" -d '{"address":"YOUR_ADDRESS"}'


4. MINING
---------
# Start mining (RPC)
./target/release/quanta start_mining <WALLET_ADDRESS> --rpc-port 7782

# Stop mining (RPC)
./target/release/quanta stop_mining --rpc-port 7782

# Mine single block (API)
curl -X POST http://localhost:3000/api/mine -H "Content-Type: application/json" -d '{"miner_address":"YOUR_ADDRESS"}'


5. API ENDPOINTS
----------------
# Get stats
curl http://localhost:3000/api/stats

# Get peer count
curl http://localhost:3000/api/peers

# Get mempool info
curl http://localhost:3000/api/mempool


6. MAINTENANCE
--------------
